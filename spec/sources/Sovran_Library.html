DataFrame Operations

data_analysis.jl 
using Sovran.DataFrame

# Load data
df = read_csv("sales_data.csv")

# Filter and transform
filtered = filter(df, row -> row.sales > 1000)
filtered.profit = filtered.revenue .- filtered.cost

# GroupBy aggregation
grouped = groupby(filtered, :region)
summary = aggregate(grouped,
    total_sales = :sales => sum,
    avg_profit = :profit => mean,
    count = :id => length
)

# Join with another table
regions = read_csv("regions.csv")
result = leftjoin(summary, regions, on=:region)

write_csv(result, "output.csv")

Physics-Informed NN

pinn_burgers.jl 
using Sovran
using Sovran.UFTOps

# Define PINN for Burgers equation
struct BurgersPINN
    network::Sequential
    viscosity::Float64
end

# PDE residual: u_t + u*u_x - ν*u_xx = 0
function pde_residual(pinn, x, t)
    xt = Variable(cat(x, t), grad=true)
    u = pinn.network(xt)
    
    backward(u)
    u_t = xt.grad[:, 2]
    u_x = xt.grad[:, 1]
    
    zero_grad(xt)
    backward(u_x)
    u_xx = xt.grad[:, 1]
    
    return u_t .+ u .* u_x .- pinn.viscosity .* u_xx
end

# Train with physics constraints
pinn = BurgersPINN(network, viscosity=0.01)
train_pinn(pinn, data, collocation)

Distributed Training

distributed_train.jl 
using Sovran.ExecRT
using Distributed

# Initialize cluster
addprocs(8)  # 8 GPU workers

@everywhere using Sovran

# Distributed data loader
loader = DistributedDataLoader(
    "/data/imagenet",
    batch_size=64
)

# Synchronous SGD with all-reduce
distributed_train(model, loader;
    epochs=100,
    lr=0.01,
    sync_strategy="all_reduce"
)

# Save checkpoint
save_model(model, "checkpoint.sjl")
